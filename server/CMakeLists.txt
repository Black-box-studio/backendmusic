cmake_minimum_required(VERSION 3.21)
project(GlassMusicServer VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Accommodate distributions which package Oracle MySQL instead of MariaDB.
if(UNIX)
    find_library(MYSQL_LIBRARIES NAMES mysqlclient mariadb)
    find_path(MYSQL_INCLUDE_DIRS NAMES mysql.h PATH_SUFFIXES mysql mariadb)
endif()
find_package(Drogon CONFIG REQUIRED)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
configure_file(../provider/storage_provider.py storage_provider.py COPYONLY)
find_package(OpenSSL 3 REQUIRED)
if(NOT TARGET SQLite3_lib)
    find_package(SQLite3 REQUIRED)
endif()
add_executable(glass-music-server main.cpp database.cpp security.cpp service.cpp storage.cpp)
target_include_directories(glass-music-server PRIVATE ${SQLITE3_INCLUDE_DIRS})
target_link_libraries(glass-music-server PRIVATE Drogon::Drogon OpenSSL::Crypto Qt6::Core ${SQLITE3_LIBRARIES})
if(MSVC)
    target_compile_options(glass-music-server PRIVATE /W4 /permissive-)
else()
    target_compile_options(glass-music-server PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS glass-music-server RUNTIME DESTINATION bin)

install(FILES ../provider/storage_provider.py DESTINATION bin)
